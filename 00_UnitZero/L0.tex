\section{Review of Differential Equations}
\subsection{Separation of variables}
General First Order ODE:
\begin{align*}
\frac{dy}{dt}+py &= g
\end{align*}
If p and g are constants we can solve this by separation of variables:

\begin{align*}
    \frac{dy}{dt} + py &= g \\
    \frac{dy}{dt} &= g - py \\
    \frac{1}{g - py} dy &= dt \\
    \intertext{Integrate both sides:}
    \int \frac{1}{g - py} dy &= \int 1 dt \\
    \intertext{\textcolor{blue}{Substitution: Let $u = g - py \implies du = -p dy$}}
    \intertext{\textcolor{blue}{$\implies dy = -\frac{1}{p}du$}}
    \int \frac{1}{u} \left( -\frac{1}{p} \right) du &= t + C \\
    -\frac{1}{p} \ln|u| &= t + C \\
    \ln|u| &= -p(t + C) \\
    \ln|u| &= -pt + C_{new} \quad \text{(where } C_{new} = -pC \text{)} \\
    |u| &= e^{-pt + C_{new}} \\
    u &= \pm e^{C_{new}} e^{-pt} \\
    \intertext{Let $C_1 = \pm e^{C_{new}}$. Substitute back $u = g - py$:}
    g - py &= C_1 e^{-pt} \\
    -py &= C_1 e^{-pt} - g \\
    y &= \frac{C_1 e^{-pt}}{-p} + \frac{-g}{-p} \\
    \boxed{y(t) = C_2 e^{-pt} + \frac{g}{p}} &\quad \text{where } C_2 = -\frac{C_1}{p}
\end{align*}

\subsubsection*{Behavior Analysis}
The solution consists of two distinct parts:
\begin{equation*}
    y(t) = \underbrace{C_2 e^{-pt}}_{\text{Transient Solution}} + \underbrace{\frac{g}{p}}_{\text{Steady State Solution}}
\end{equation*}
\begin{itemize}
    \item \textbf{Transient:} If $p > 0$, this term decays to zero as $t \to \infty$.
    \item \textbf{Steady State:} The value $y(t)$ approaches $\frac{g}{p}$ as time goes on.
    \item \textbf{Stability:} If $p < 0$, the exponential term grows to infinity (Unstable).
\end{itemize}

\subsubsection*{Solving for the Initial Condition}
Given an initial value $y(0) = y_0$, we can solve for $C_2$:
\begin{align*}
    y(0) &= C_2 e^{-p(0)} + \frac{g}{p} \\
    y_0 &= C_2 (1) + \frac{g}{p} \\
    C_2 &= y_0 - \frac{g}{p}
\end{align*}
Substituting this back into the general solution gives the specific solution:
\begin{equation*}
    \boxed{y(t) = \left( y_0 - \frac{g}{p} \right) e^{-pt} + \frac{g}{p}}
\end{equation*}
Remember, this is for a LINEAR 1ST ORDER ODE WITH CONSTANT COEFFICIENTS.

\begin{align*}
    \tikzmarknode{eq}{\frac{dy}{dt}} + p(t)y &= \tikzmarknode{qt}{q(t)}

Rewrite to match the form of :
\begin{align*}
    \tikzmarknode{matchE}{\frac{d}{dt}}\left(\mu(t)y\right)&=\mu(t)q(t)\\
    \int\frac{d}{dt}\left(\mu(t)y\right)dt'&=\int_{0}^{T}\mu(t)q(t)dt'\\
    \mu(t)y(t)-\mu(0)y(0)&=\int_{0}^{T}\mu(t')q(t')dt'\\
\mu(t)y(t)&=\int_{0}^{T}\mu(t')q(t')dt'+\mu(0)y(0)\\

\end{align*}
To find $\mu$, we match both sides of the equation.
\begin{align*}
    \tikzmarknode{matchS}{\frac{d}{dt}}\Bigg(\mu(t)y\Bigg)&=\frac{d\mu}{dt}y+\mu \frac{dy}{dt} \tag{$\bigstar$}

Multiply the original equation by $\mu$:
\begin{align*}

    \mu\frac{dy}{dt}+\mu p(t)y&=\mu q(t) \tag{$\Delta$}
\end{align*}
Now we can compare $\bigstar$ and $\Delta$

\begin{align*}
    \frac{d\mu}{dt}&=\mu p(t)\\
    \frac{1}{\mu}d\mu&=p(t)dt\\
    \mu&=\boxed{Ce^{\int p(t)dt}}
\end{align*}

\subsubsection*{Note on the Constant $C$}
\begin{align*}
    &\text{We can choose } C=1 \text{ because any constant cancels out.} \\
    &\text{why:} \\
    &\text{Let } \mu = C e^{\int p(t)dt}. \text{ Substitute into general solution:} \\
    y(t) &= \frac{1}{C e^{\int p dt}} \Bigg[ \int C e^{\int p dt} q(t) dt + C_{int} \Bigg] \\
    &\text{Factor out } C \text{ from the integral:} \\
    y(t) &= \frac{1}{C e^{\int p dt}} \cdot C \Bigg[ \int e^{\int p dt} q(t) dt + \frac{C_{int}}{C} \Bigg] \\
    &\text{Cancel the } C \text{ terms:} \\
    y(t) &= \frac{1}{e^{\int p dt}} \Bigg[ \int e^{\int p dt} q(t) dt + C_{new} \Bigg] \\
   &\text{ The constant } C \text{ in } \mu \text{ is redundant.}
\end{align*}

\subsubsection*{Big Must: Standard Form}
\begin{align*}
    &\text{The derivation assumes the coefficient of } \frac{dy}{dt} \text{ is } \mathbf{1}. \\
    &\text{If given a general form:} \\
    &a(t)\frac{dy}{dt} + b(t)y = g(t) \\
    &\text{You \textbf{must} divide by } a(t) \text{ first:} \\
    &\frac{dy}{dt} + \frac{b(t)}{a(t)}y = \frac{g(t)}{a(t)} \\
    &\text{Identify your terms:} \\
    &p(t) = \frac{b(t)}{a(t)}, \quad q(t) = \frac{g(t)}{a(t)}
\end{align*}

\subsubsection*{Algorithm Summary}
\begin{align*}
    1. \quad &\textbf{Standardize:} \\
    &\text{Ensure } \frac{dy}{dt} \text{ has a coefficient of 1.} \\
    2. \quad &\textbf{Identify:} \\
    &\text{Find } p(t) \text{ and } q(t). \\
    3. \quad &\textbf{Calculate } \boldsymbol{\mu}: \\
    &\mu(t) = e^{\int p(t)dt} \quad (\text{ignore } +C). \\
    4. \quad &\textbf{Solve:} \\
    &y(t) = \frac{1}{\mu(t)} \Bigg[ \int \mu(t)q(t) dt + C \Bigg].
\end{align*}

\subsection{2nd Order Linear Differential Equations}
A general form of a 2nd order ODE:
\begin{align*}
    \frac{d^2y}{dt^2}&= f\Bigg(t,y,\frac{dy}{dt}\Bigg)\\
    &\textbf{or}\\
    F\Bigg(t,y,\frac{dy}{dt},\frac{d^2y}{dt^2}\Bigg)&=0
\end{align*}

\subsection*{A quick note of Implicit, Explicit and Homogeneity}
When we say $F\Bigg(x,y,PB,Jelly,t,\frac{dy}{dt},\frac{d^2y}{dt^2}\Bigg)$, the \textbf{Container} F holds all combinations of said \\
$x,y,PB,Jelly,t,\frac{dy}{dt},\frac{d^2y}{dt^2}$ that add add to \textcolor{red}{0}.
Please note, this doesn't mean its \textcolor{red}{Homogeneous} in how its presented here. More on this in a little.\\
\\
For the other form:
\begin{align*}
    \frac{d^2y}{dt^2}&= f\Bigg(t,y,\frac{dy}{dt}\Bigg)
\end{align*}
Here we are saying, "if you give me current time, position and velocity, Ill be able to calculate acceleration"
\\
\\
Back to \textbf{Homogeneous or not}.\\
In the form $F\Bigg(t,y,\frac{dy}{dt},\frac{d^2y}{dt^2}\Bigg)=0$ we have only moved all things in the container to one side. Using my ridiculous example from before,\\
\\
Imagine an Implicit Function $F$ representing the ingredients of a lunch equation:
\begin{align*}
    F\Bigg(x, y, \text{PB}, \text{Jelly}, t, \frac{dy}{dt}, \frac{d^2y}{dt^2}\Bigg) &= 0
\end{align*}
This is \textbf{Implicit} because the ingredients are all mixed together in the bucket $F$.

To check if it is \textbf{Homogeneous}, we try to isolate the sandwich mechanics ($y, y', y''$) from the external ingredients:
\begin{align*}
    \underbrace{\frac{d^2y}{dt^2} + p(t)\frac{dy}{dt} + q(t)y}_{\text{The Bread}} &= \underbrace{\text{PB} + \text{Jelly}}_{\text{External Forcing Function } f(t)}
\end{align*}
\begin{itemize}
    \item If $\text{PB} = 0$ and $\text{Jelly} = 0$, the equation is \textbf{Homogeneous} (Just Bread).
    \item If $\text{PB} \neq 0$ or $\text{Jelly} \neq 0$, the equation is \textbf{Non-Homogeneous} (A Sandwich exists).
\end{itemize}

\subsection*{The Linear Case}
\begin{align*}
    &\frac{d^2y}{dt^2}=g(t)-p(t)\frac{dy}{dt}-q(t)y\\
    \intertext{if $g(t)=0$;}\\
    &\frac{d^2y}{dt^2}+p(t)\frac{dy}{dt}+q(t)y=g(t)\\        &\frac{d^2y}{dt^2}+p(t)\frac{dy}{dt}+q(t)y=0\\
    \intertext{This is now homogeneous and we can move forward with a characteristic equation.}\\
\end{align*}
\subsection*{The Characteristic Equation}
\begin{align*}
    \frac{d^2y}{dt^2}+p(t)\frac{dy}{dt}+q(t)y&=0\\
    \intertext{change the $\frac{d}{dt}$ terms to r's}\\
    r^2y+p(t)ry+q(t)r^0y=0\\
    \intertext{we can now change the coefficients to manageable variables}\\
    \Bigg[ar^2+br+c\Bigg]y=0\\
    \underbrace{ar^2+br+c=0}_{\text{Characteristic equation}}\\
    \intertext{$\rightarrow$ find the roots $r_1$ and $r_2$ of the characteristic}\\
\end{align*}

\begin{align*}
    (r-r_1)(r-r_2)y&=0\\
    \intertext{This equation is satisfied when:}\\
    (r-r_1)y=0 \quad &\textbf{or} \quad (r-r_2)y=0\\
\end{align*}
\noindent
\begin{minipage}[t]{0.45\textwidth}
    
    \begin{align*}
        (r-r_1)y &= 0 \\
        (\frac{d}{dt}-r_1)y&=0\\
        \frac{dy}{dt} &= r_1 y_1 \\
        y_1 &= C_1 e^{r_1 t}
    \end{align*}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.45\textwidth}
    
    \begin{align*}
        (r-r_2)y &= 0 \\
        (\frac{d}{dt}-r_2)y&=0\\
        \frac{dy}{dt} &= r_2 y_2 \\
        y_2 &= C_2 e^{r_2 t}
    \end{align*}
\end{minipage}
So the general solution for this would be:
\begin{align*}
    y=C_1e^{r_1t}+C_2e^{r_2t}
\end{align*}
$C_1$ and $C_2$ will both be found by initial conditions.
\subsection*{Example:}
\begin{align*}
    y''+4y'+4y&=0 \tag{original}\\
    r^2+4r+4&=0\\
    (r+2)(r+2)&=0\\
    (r+2)^2&=0
\end{align*}
We know that one linear independence is of the form:

\begin{align*}
    y_1&=C_1e^{-2t}\\
    &\intertext{for the other solution we can assume:}\\
    y_2&=\nu(t)e^{-2t}\\

    &=\nu'(t)e^{-2t}-2\nu(t)e^{-2t}\\
    y_2''&=\frac{d}{dt}\nu'(t)e^{-2t}-2\nu(t)\frac{d}{dt}e^{-2t}\\
    &=\nu''(t)e^{-2t}-4\nu'(t)e^{-2t}+4\nu(t)e^{-2t}\\
    &\intertext{now into the original:}\\
    y_2''+4y_2'+4y&=0
\end{align*}
\begin{align*}
\nu''(t)e^{-2t}-4\nu'(t)e^{-2t}+4\nu(t)e^{-2t}+4(\nu'(t)e^{-2t}-2\nu(t)e^{-2t})+4(\nu(t)e^{-2t})&=0\\

v(t)&=C_1t +C_2\\

\tikzmarknode{nuS}{\nu(t)}&=t

\subsection*{Note on Complex Roots}
If the characteristic equation $ar^2+br+c=0$ yields complex roots:
\begin{align*}
    r_{1,2} &= \lambda \pm i\mu
\end{align*}
The general solution takes the form:
\begin{align*}
    y(t) &= e^{\lambda t} \Big( C_1 \cos(\mu t) + C_2 \sin(\mu t) \Big)
\end{align*}
\textbf{Why?} This comes from Euler's Identity:
\begin{align*}
    e^{i\theta} &= \cos \theta + i \sin \theta
\end{align*}
Applying this to our root $r = \lambda + i\mu$:
\begin{align*}
    e^{(\lambda + i\mu)t} &= e^{\lambda t} \cdot e^{i(\mu t)} \\
    &= e^{\lambda t} (\cos \mu t + i \sin \mu t)
\end{align*}
(The imaginary terms form the oscillation $\mu$, and the real part $\lambda$ forms the growth/decay envelope).

\subsection{Non-Homogeneous 2nd order ODE}
\subsection*{Method of Undetermined Coefficients}
Example:
\begin{align*}
    y''-3y'-4y &= 3e^{2t} \\
    \intertext{Step 1: Find the homogeneous ($y_h$) solution via the characteristic eqn:}
    r^2-3r-4 &= 0 \\
    (r-4)(r+1) &= 0 \\
    r &= 4, -1 \\
    y_h(t) &= C_1e^{4t} + C_2e^{-t}
\end{align*}

Now we make an \tikzmarknode{anz}{\textbf{Ansatz}} (guess) for the particular solution $y_p(t)$ based on the forcing function $3e^{2t}$.
\\
It needs to satisfy the ODE:
\[ y_p''(t) - 3y_p'(t) - 4y_p(t) = 3e^{2t} \]

Since the forcing function is an exponential, we guess a matching form:
\begin{align*}
    y_p(t) &= Ae^{2t} \\
    y_p'(t) &= 2Ae^{2t} \\
    y_p''(t) &= 4Ae^{2t}
\end{align*}
Substitute these back into the original ODE:
\begin{align*}
    4Ae^{2t} - 3(2Ae^{2t}) - 4(Ae^{2t}) &= 3e^{2t} \\
    (4 - 6 - 4)Ae^{2t} &= 3e^{2t} \\
    -6A &= 3 \\
    A &= -\frac{1}{2}
\end{align*}
So the general solution is:
\begin{align*}
    y(t) &= y_h(t) + y_p(t) \\
    \boxed{y(t) = C_1e^{4t} + C_2e^{-t} - \frac{1}{2}e^{2t}}
\end{align*}

What happens if we change things a little??

lets say we instead had,
\begin{align*}
    y''-3y'-4y&=3\tikzmarknode{ldE}{e^{-t}}\\
\end{align*}
The $y_c$ part would still be $C_1e^{4t}+C_2\textcolor{red}{\tikzmarknode{ldS}{e^{-t}}}$

    \draw[->, >=Latex, red, thick, rounded corners] (ldS.east) 
        -- ++(2.8, 0) coordinate(turnPoint1)      
        -- (turnPoint1 |- ldE.south) ;  
\end{tikzpicture}
Notice that the $y_c$ we chose is no longer linearly independent from the $y_p$ we are going to have. So what we need to do now is change the ansatz to $Ate^{2t}$. \\
We need $y_p$ to be \textbf{Linearly Independent} from the homogeneous solution $y_c$.

\begin{itemize}
    \item \textbf{The Conflict:} If our guess $y_p$ appears in $y_c$, then $y_p$ is \textbf{Linearly Dependent} (it is just a copy of the natural response). Plugging it into the ODE will yield $0 = g(t)$, which is impossible.
    \item \textbf{The Fix:} Multiplying by $t$ breaks this dependency. The functions $e^{-t}$ and $te^{-t}$ are Linearly Independent, allowing the math to balance.
\end{itemize}

\subsubsection*{Resonance}
\textbf{Important:} If any term in your guess $y_p(t)$ is \textit{already} a part of the homogeneous solution $y_c(t)$, you must multiply your guess by $t^s$, where $s$ is the smallest integer $(1, 2, \dots)$ that makes the term unique.

\begin{itemize}
    \item \textbf{Example:} $y'' - y' = 3$. 
    \begin{itemize}
        \item Homogeneous: $y_c = c_1 + c_2 e^t$.
        \item Naive Guess for $g(t)=3$: $y_p = A$.
        \item \textbf{Problem:} The constant $A$ is absorbed by $c_1$.
        \item \textbf{Fix:} Multiply by $t \implies \boxed{y_p = At}$.
    \end{itemize}
\end{itemize}
\subsection*{Example}
\begin{align*}
    y'' - 2y' + y &= e^x + x
\end{align*}
\textbf{Step 1: Homogeneous Solution}
\begin{align*}
    r^2 - 2r + 1 &= 0 \\
    (r-1)^2 &= 0 \quad \implies r = 1, 1 \text{ (Repeated!)} \\
    y_h &= C_1 e^x + \underbrace{C_2 x e^x}_{\text{Repeated root rule}}
\end{align*}

\textbf{Step 2: Particular Solution ($y_p$)}
The forcing function is $g(x) = e^x + x$. We split this into two parts:
\begin{enumerate}
    \item \textbf{Part A ($e^x$):}
    \begin{itemize}
        \item Initial guess: $A e^x$.
        \item Conflict: Overlaps with $C_1 e^x$. $\rightarrow$ Try $Ax e^x$.
        \item Conflict: Overlaps with $C_2 x e^x$. $\rightarrow$ Try $Ax^2 e^x$.
        \item \textbf{Final Guess A:} $y_{p1} = A x^2 e^x$
    \end{itemize}
    
    \item \textbf{Part B ($x$):}
    \begin{itemize}
        \item This is a degree 1 polynomial.
        \item \textbf{Final Guess B:} $y_{p2} = Bx + C$
    \end{itemize}
\end{enumerate}

\textbf{Combined Guess:}
\begin{align*}
    y_p &= A x^2 e^x + Bx + C \\
    y_p' &= A(2x e^x + x^2 e^x) + B \\
    y_p'' &= A(2e^x + 2xe^x + 2xe^x + x^2e^x) = A e^x(x^2 + 4x + 2)
\end{align*}
Substitute these into the original ODE ($y'' - 2y' + y = e^x + x$):
\begin{align*}
    \Big[ Ae^x(x^2 + 4x + 2) \Big] - 2 \Big[ Ae^x(x^2 + 2x) + B \Big] + \Big[ Ax^2e^x + Bx + C \Big] &= e^x + x
\end{align*}

\subsection*{Step 4: Solving for Coefficients}
Group the terms by type ($e^x$ terms, $x$ terms, and constants):
\begin{align*}
    \textbf{Exp terms:} \quad & Ae^x \Big[ (x^2 + 4x + 2) - 2(x^2 + 2x) + x^2 \Big] = Ae^x[2] \\
    \textbf{Poly terms:} \quad & Bx + (C - 2B)
\end{align*}
Set equal to the right hand side ($1e^x + 1x + 0$):
\begin{align*}
    2Ae^x &= 1e^x \implies 2A = 1 \implies \boxed{A = \frac{1}{2}} \\
    Bx &= 1x \implies \boxed{B = 1} \\
    C - 2B &= 0 \implies C - 2(1) = 0 \implies \boxed{C = 2}
\end{align*}

\subsection*{Step 5: General Solution}
Combine $y_h$ and the solved $y_p$:
\begin{align*}
    y(x) &= y_h + y_p \\
   &\boxed{y(x) = C_1 e^x + C_2 x e^x + \frac{1}{2}x^2 e^x + x + 2}
\end{align*}
\subsection{Variation of Parameters (Lagrange)}
\textbf{Use Case:} When the forcing function $r(x)$ is not amenable to Undetermined Coefficients (e.g., $\sec x, \tan x, \ln x$), or coefficients are not constant.

\begin{align*}
    \intertext{\textbf{Standard Form Assumption:}}
    y'' + f(x)y' + g(x)y &= r(x) \\
    \intertext{\textbf{Warning:} The coefficient of $y''$ must be 1. If it is $x^2y''$, divide by $x^2$ first!}
\end{align*}

\subsubsection*{Step 1: The Homogeneous Foundation}
First, find the complementary solution ($y_h$) satisfying $y''+fy'+gy=0$:
\[ y_h(x) = C_1 y_1(x) + C_2 y_2(x) \]

\subsubsection*{Step 2: Variation of Constants}
We replace the constants $C_1, C_2$ with unknown functions $u(x), v(x)$:
\[ y_p(x) = u(x)y_1(x) + v(x)y_2(x) \]

\subsubsection*{Step 3: Deriving the System}
To find $u$ and $v$, we need two equations.

\textbf{1. The Constraint (Avoiding $u'', v''$):}
Differentiating $y_p$ initially gives terms with $u'$ and $v'$. To simplify, we choose:
\[ \boxed{u'y_1 + v'y_2 = 0} \]
This simplifies the first derivative to: $y_p' = u y_1' + v y_2'$.

\textbf{2. The Second Derivative \& Substitution:}
Differentiate $y_p'$ again to get $y_p''$:
\[ y_p'' = (u'y_1' + u y_1'') + (v'y_2' + v y_2'') \]
Substitute $y_p, y_p', y_p''$ into the ODE ($y'' + Py' + Qy = r$):
\[
    \underbrace{[(u'y_1' + u y_1'') + (v'y_2' + v y_2'')]}_{y''} + P\underbrace{[u y_1' + v y_2']}_{y'} + Q\underbrace{[u y_1 + v y_2]}_{y} = r
\]
Group by $u$ and $v$. Since $y_1, y_2$ are homogeneous solutions, their terms sum to zero:
\[
    u(\underbrace{y_1'' + Py_1' + Qy_1}_{0}) + v(\underbrace{y_2'' + Py_2' + Qy_2}_{0}) + (u'y_1' + v'y_2') = r
\]
This leaves the second equation:
\[ \boxed{u'y_1' + v'y_2' = r(x)} \]

\subsubsection*{Step 4: Solve with Wronskian}
We solve the system for $u', v'$ using Cramer's Rule:
\begin{align*}
    W &= \begin{vmatrix} y_1 & y_2 \\ y_1' & y_2' \end{vmatrix} = y_1 y_2' - y_2 y_1'
\end{align*}
The formulas for the derivatives are:
\begin{align*}
    u'(x) &= \frac{-y_2(x) r(x)}{W}, \quad v'(x) = \frac{y_1(x) r(x)}{W}
\end{align*}

\subsubsection*{Step 5: Integrate}
Integrate $u', v'$ to find the functions, then plug back into the Ansatz:
\[ y_p(x) = \left( \int u'(x) dx \right)y_1(x) + \left( \int v'(x) dx \right)y_2(x) \]
